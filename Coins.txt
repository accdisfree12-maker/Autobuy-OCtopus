import discord
from discord import app_commands
from discord.ext import commands
import json
import os
import datetime
import random

# --- CONFIGURATION ---
ALLOWED_CHANNEL_ID = 1540870304459530341  # Channel ID where 'pr' and 'ct' are allowed
ADDCREDIT_ADMIN_IDS = [ Put ur id here ]  # Specific IDs allowed for /addcredit
DB_FILE = "database.json"

intents = discord.Intents.default()
intents.message_content = True  # Required to read text messages like "pr" and "ct"

bot = commands.Bot(command_prefix="!", intents=intents)

# --- HELPER FUNCTIONS ---
def is_admin(interaction: discord.Interaction) -> bool:
    """Check if the user has Discord Administrator permissions."""
    return interaction.user.guild_permissions.administrator

def can_add_credit(interaction: discord.Interaction) -> bool:
    """Check if the user ID is explicitly allowed to add credit."""
    return interaction.user.id in ADDCREDIT_ADMIN_IDS

def parse_amount(amount_str: str) -> int:
    """Converts inputs like 1k, 5k, 2.5k into integers (e.g., 5k -> 5000)."""
    amount_str = str(amount_str).lower().strip()
    multiplier = 1
    if amount_str.endswith("k"):
        multiplier = 1000
        amount_str = amount_str[:-1]
    elif amount_str.endswith("m"):
        multiplier = 1000000
        amount_str = amount_str[:-1]
    try:
        val = float(amount_str) * multiplier
        return int(val)
    except ValueError:
        return -1

def load_db():
    if not os.path.exists(DB_FILE):
        with open(DB_FILE, "w") as f:
            json.dump({"users": {}, "logs": []}, f, indent=4)
    with open(DB_FILE, "r") as f:
        return json.load(f)

def save_db(data):
    with open(DB_FILE, "w") as f:
        json.dump(data, f, indent=4)

def get_user_data(db, user_id):
    user_str = str(user_id)
    if user_str not in db["users"]:
        db["users"][user_str] = {
            "credit": 0,
            "last_daily": None,
            "cmd_count": 0
        }
    if "cmd_count" not in db["users"][user_str]:
        db["users"][user_str]["cmd_count"] = 0
    return db["users"][user_str]

def get_level(cmd_count: int) -> int:
    """Increases Level every 25 commands."""
    return 1 + (cmd_count // 25)

def increment_cmd_count(user_id: int):
    db = load_db()
    user_data = get_user_data(db, user_id)
    user_data["cmd_count"] += 1
    save_db(db)

def build_profile_embed(user: discord.User, user_data: dict, has_admin: bool) -> discord.Embed:
    lvl = get_level(user_data["cmd_count"])
    embed = discord.Embed(
        title=f"👤 Profile of {user.name}",
        color=0xFFD700 if has_admin else 0x0099FF
    )
    avatar_url = user.display_avatar.url if hasattr(user, 'display_avatar') else user.default_avatar.url
    embed.set_thumbnail(url=avatar_url)
    embed.add_field(name="💵 Credits", value=f"{user_data['credit']}", inline=True)
    embed.add_field(name="📊 Level", value=f"Lvl {lvl} ({user_data['cmd_count']} cmds)", inline=True)
    embed.add_field(name="🛡️ Role", value="Administrator" if has_admin else "Member", inline=True)
    return embed

# --- EVENTS ---
@bot.event
async def on_ready():
    print(f"🤖 Bot connected as {bot.user.name}")
    try:
        synced = await bot.tree.sync()
        print(f"✅ Synced {len(synced)} slash command(s).")
    except Exception as e:
        print(f"❌ Sync error: {e}")

@bot.event
async def on_app_command_completion(interaction: discord.Interaction, command):
    increment_cmd_count(interaction.user.id)

@bot.event
async def on_message(message: discord.Message):
    if message.author.bot:
        return

    # Text triggers restricted to ALLOWED_CHANNEL_ID
    if message.channel.id == ALLOWED_CHANNEL_ID:
        content = message.content.strip().lower()
        if content == "pr":
            increment_cmd_count(message.author.id)
            db = load_db()
            user_data = get_user_data(db, message.author.id)
            has_admin = message.author.guild_permissions.administrator if message.guild else False
            embed = build_profile_embed(message.author, user_data, has_admin)
            await message.channel.send(embed=embed)
            return

        elif content == "ct":
            increment_cmd_count(message.author.id)
            db = load_db()
            user_data = get_user_data(db, message.author.id)
            await message.channel.send(f"💰 You have **{user_data['credit']}** credits.")
            return

    await bot.process_commands(message)

@bot.tree.command(name="credit", description="Display your current credit balance.")
async def credit(interaction: discord.Interaction):
    db = load_db()
    user_data = get_user_data(db, interaction.user.id)
    await interaction.response.send_message(f"💰 You have **{user_data['credit']}** credits.", ephemeral=True)

@bot.tree.command(name="profile", description="View your profile information.")
async def profile(interaction: discord.Interaction):
    db = load_db()
    user_data = get_user_data(db, interaction.user.id)
    has_admin = is_admin(interaction)
    embed = build_profile_embed(interaction.user, user_data, has_admin)
    await interaction.response.send_message(embed=embed)

@bot.tree.command(name="pay", description="Transfer credits to another user.")
@app_commands.describe(user="Select the user to transfer credits to", amount="The amount (e.g., 500, 1k)")
async def pay(interaction: discord.Interaction, user: discord.User, amount: str):
    parsed_amount = parse_amount(amount)
    if parsed_amount <= 0:
        return await interaction.response.send_message("❌ Invalid amount.", ephemeral=True)

    if user.id == interaction.user.id:
        return await interaction.response.send_message("❌ You cannot send credits to yourself!", ephemeral=True)

    db = load_db()
    sender = get_user_data(db, interaction.user.id)

    if sender["credit"] < parsed_amount:
        return await interaction.response.send_message("❌ You do not have enough credits.", ephemeral=True)

    receiver = get_user_data(db, user.id)

    sender["credit"] -= parsed_amount
    receiver["credit"] += parsed_amount

    now_str = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    db["logs"].append({
        "sender": interaction.user.id,
        "receiver": user.id,
        "amount": parsed_amount,
        "date": now_str
    })

    save_db(db)
    await interaction.response.send_message(f"✅ You successfully transferred **{parsed_amount}** credits to {user.mention}.")

@bot.tree.command(name="daily", description="Claim your daily credit reward (100-500).")
async def daily(interaction: discord.Interaction):
    db = load_db()
    user_data = get_user_data(db, interaction.user.id)
    now = datetime.datetime.now()

    if user_data["last_daily"]:
        last_daily = datetime.datetime.fromisoformat(user_data["last_daily"])
        delta = now - last_daily
        if delta.total_seconds() < 86400:
            remaining = 86400 - delta.total_seconds()
            hours = int(remaining // 3600)
            minutes = int((remaining % 3600) // 60)
            return await interaction.response.send_message(f"⌛ Please come back in **{hours}h and {minutes}m** for your daily reward.", ephemeral=True)

    reward = random.randint(100, 500)
    user_data["credit"] += reward
    user_data["last_daily"] = now.isoformat()
    save_db(db)

    await interaction.response.send_message(f"🎉 You received your random daily reward of **{reward}** credits!")

@bot.tree.command(name="addcredit", description="[Authorized Admins] Add credits to a user (Max 5k).")
@app_commands.describe(user="Select the user", amount="Amount (e.g., 1k, 5k, 2500)")
async def addcredit(interaction: discord.Interaction, user: discord.User, amount: str):
    if not can_add_credit(interaction):
        return await interaction.response.send_message("🚫 You do not have permission to use this command.", ephemeral=True)

    parsed_amount = parse_amount(amount)
    if parsed_amount <= 0:
        return await interaction.response.send_message("❌ Invalid amount.", ephemeral=True)

    if parsed_amount > 5000:
        return await interaction.response.send_message("❌ The maximum allowed amount to add is **5k** (5000).", ephemeral=True)

    db = load_db()
    target = get_user_data(db, user.id)
    target["credit"] += parsed_amount
    save_db(db)

    await interaction.response.send_message(f"✅ Added **{parsed_amount}** credits to {user.mention}.")

@bot.tree.command(name="resetcredit", description="[Admin] Reset a user's credit balance to zero.")
@app_commands.describe(user="Select the user")
async def resetcredit(interaction: discord.Interaction, user: discord.User):
    if not is_admin(interaction):
        return await interaction.response.send_message("🚫 This command is reserved for server Administrators.", ephemeral=True)

    db = load_db()
    target = get_user_data(db, user.id)
    target["credit"] = 0
    save_db(db)

    await interaction.response.send_message(f"🔄 Reset credit balance of {user.mention} to **0**.")

@bot.tree.command(name="logs", description="[Admin] View transaction history logs.")
@app_commands.describe(user="(Optional) Select a user to filter logs")
async def logs(interaction: discord.Interaction, user: discord.User = None):
    if not is_admin(interaction):
        return await interaction.response.send_message("🚫 This command is reserved for server Administrators.", ephemeral=True)

    db = load_db()
    logs_list = db["logs"]

    if user:
        logs_list = [l for l in logs_list if l["sender"] == user.id or l["receiver"] == user.id]

    if not logs_list:
        return await interaction.response.send_message("📋 No transaction logs found.", ephemeral=True)

    recent_logs = logs_list[-10:]
    recent_logs.reverse()

    embed = discord.Embed(title="📜 Transaction Logs", color=0x2F3136)
    desc = ""
    for log in recent_logs:
        desc += f"⏰ **{log['date']}**\n📤 <@{log['sender']}> ➔ 📥 <@{log['receiver']}> : **{log['amount']}** credits\n---\n"

    embed.description = desc
    await interaction.response.send_message(embed=embed, ephemeral=True)

@bot.tree.command(name="userlevel", description="[Admin] View a user's level and executed command count.")
@app_commands.describe(user="Select the user")
async def userlevel(interaction: discord.Interaction, user: discord.User):
    if not is_admin(interaction):
        return await interaction.response.send_message("🚫 This command is reserved for server Administrators.", ephemeral=True)

    db = load_db()
    user_data = get_user_data(db, user.id)
    lvl = get_level(user_data["cmd_count"])

    await interaction.response.send_message(
        f"📊 **Stats for {user.mention}**:\n• **Level**: Lvl {lvl}\n• **Commands executed**: {user_data['cmd_count']}",
        ephemeral=True
    )
