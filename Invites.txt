import os
import json
import discord
from discord import app_commands
from discord.ext import commands
from datetime import datetime

DB_FILE = "invite_tracker.json"

def load_data():
    if os.path.exists(DB_FILE):
        try:
            with open(DB_FILE, "r") as f:
                return json.load(f)
        except:
            return {}
    return {}

def save_data(data):
    with open(DB_FILE, "w") as f:
        json.dump(data, f, indent=4)

intents = discord.Intents.default()
intents.members = True
intents.guilds = True
intents.invites = True

class InviteBot(commands.Bot):
    def __init__(self):
        super().__init__(command_prefix="!", intents=intents)
        self.invites_cache = {}
        self.data = load_data()

    async def setup_hook(self):
        await self.tree.sync()

bot = InviteBot()

@bot.event
async def on_ready():
    print(f"Logged in as {bot.user}")
    for guild in bot.guilds:
        try:
            bot.invites_cache[guild.id] = await guild.invites()
            g_id = str(guild.id)
            if g_id not in bot.data:
                bot.data[g_id] = {}
        except Exception:
            pass

@bot.event
async def on_member_join(member):
    guild = member.guild
    try:
        new_invites = await guild.invites()
        old_invites = bot.invites_cache.get(guild.id, [])
        
        used_invite = None
        for new_inv in new_invites:
            for old_inv in old_invites:
                if new_inv.code == old_inv.code and new_inv.uses > old_inv.uses:
                    used_invite = new_inv
                    break
            if used_invite:
                break
                
        bot.invites_cache[guild.id] = new_invites
        
        g_id = str(guild.id)
        if g_id not in bot.data:
            bot.data[g_id] = {}
            
        if used_invite and used_invite.inviter:
            inviter_id = str(used_invite.inviter.id)
            if inviter_id not in bot.data[g_id]:
                bot.data[g_id][inviter_id] = {"members": []}
            
            exists = False
            for m in bot.data[g_id][inviter_id]["members"]:
                if m["user_id"] == member.id:
                    exists = True
                    m["left"] = False
                    m["join_time"] = datetime.utcnow().strftime("%Y-%m-%d %H:%M")
                    break
            
            if not exists:
                member_info = {
                    "user_name": member.name,
                    "user_id": member.id,
                    "join_time": datetime.utcnow().strftime("%Y-%m-%d %H:%M"),
                    "left": False
                }
                bot.data[g_id][inviter_id]["members"].append(member_info)
            
            save_data(bot.data)
    except Exception as e:
        print(f"Error join: {e}")

@bot.event
async def on_member_remove(member):
    guild = member.guild
    try:
        g_id = str(guild.id)
        if g_id in bot.data:
            for inviter_id, info in bot.data[g_id].items():
                for m in info["members"]:
                    if m["user_id"] == member.id:
                        m["left"] = True
                        save_data(bot.data)
    except Exception as e:
        print(f"Error remove: {e}")

async def send_invite_embed(target, guild, ctx_or_interaction, is_slash=True):
    g_id = str(guild.id)
    target_id = str(target.id)
    
    api_uses = 0
    try:
        g_invites = await guild.invites()
        for inv in g_invites:
            if inv.inviter and inv.inviter.id == target.id:
                api_uses += inv.uses
    except:
        pass

    members_list = []
    if g_id in bot.data and target_id in bot.data[g_id]:
        members_list = bot.data[g_id][target_id]["members"]
        
    total_joined = len(members_list)
    total_left = sum(1 for m in members_list if m["left"])
    active_count = max(0, total_joined - total_left)
        
    embed = discord.Embed(
        title=f"📊 Invite Stats for {target.name}",
        color=discord.Color.blue()
    )
    embed.add_field(name="📥 Total Joins", value=str(total_joined), inline=True)
    embed.add_field(name="📤 Leaves", value=str(total_left), inline=True)
    embed.add_field(name="✅ Active Invites", value=str(active_count), inline=True)
    
    if members_list:
        history_text = ""
        # Akhir 5 nass dakhlo mtagiin b @user
        for m in members_list[-5:]:
            status = "❌ Left" if m["left"] else "✅ Active"
            history_text += f"• <@{m['user_id']}> | Joined: {m['join_time']} | {status}\n"
        embed.add_field(name="👥 Invited Users History:", value=history_text, inline=False)
    else:
        embed.add_field(name="👥 Invited Users History:", value="No users invited yet.", inline=False)
        
    if is_slash:
        await ctx_or_interaction.response.send_message(embed=embed, ephemeral=False)
    else:
        await ctx_or_interaction.send(embed=embed)

@bot.tree.command(name="invite", description="Check invite statistics")
async def slash_invite(interaction: discord.Interaction, member: discord.Member = None):
    target = member or interaction.user
    await send_invite_embed(target, interaction.guild, interaction, is_slash=True)

@bot.command(name="invite")
async def prefix_invite(ctx, member: discord.Member = None):
    target = member or ctx.author
    await send_invite_embed(target, ctx.guild, ctx, is_slash=False)

bot.run(os.getenv("DISCORD_TOKEN"))
